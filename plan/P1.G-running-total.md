## Contract
```ts
// src/lib/totals.ts
import type { Expense } from './expense';
export function totalAmount(expenses: Expense[]): number;
// Sum of expense.amount. Empty array → 0. Pure.
```
```ts
// src/components/RunningTotal.tsx
import type { Expense } from '../lib/expense';
interface RunningTotalProps {
  expenses: Expense[];
}
export function RunningTotal(props: RunningTotalProps): JSX.Element;
// Renders the total, currency-formatted ($1,234.50), via totalAmount().
// Use a labelled element, e.g. text "Total" + the amount, so it is queryable
// by accessible name. Pure presentational.
```
App wiring (done by main agent): render `<RunningTotal expenses={expenses} />` in the
header area near the <h1>.

## Tests
- `totalAmount`: empty → 0; single expense; multiple expenses summed; floating-point
  amounts (e.g. 0.1 + 0.2 family — assert with `toBeCloseTo` or rounded).
- `RunningTotal`: renders the currency-formatted total for a given expenses array;
  renders $0.00 for an empty array.
