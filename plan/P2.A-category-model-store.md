## Contracts

### src/db/db.ts — shared DB module
```ts
// DB 'expense-tracker', VERSION 2. onupgradeneeded creates BOTH object stores
// if absent: 'expenses' (keyPath 'id') and 'categories' (keyPath 'id').
export function openDb(): Promise<IDBDatabase>
export function withStore<T>(
  storeName: string,
  mode: IDBTransactionMode,
  fn: (store: IDBObjectStore) => IDBRequest<T>,
): Promise<T>
// withStore: open db, run one transaction on storeName, resolve on tx.oncomplete
// with request.result, reject on tx.onerror/onabort, db.close() in finally.
// (This is exactly the existing expenseStore withStore, generalised over storeName.)
```

### src/lib/category.ts
```ts
export interface Category { id: string; name: string; color: string }
export interface CategoryInput { name: string; color: string }

// Validates: name non-empty after trim; color matches /^#[0-9a-fA-F]{6}$/.
// Trims name. Generates id via crypto.randomUUID(). Throws Error on invalid.
export function createCategory(input: CategoryInput): Category

// A few sensible seed categories (CategoryInput[]), e.g. Food / Transport /
// Housing / Entertainment / Other with distinct hex colors.
export const DEFAULT_CATEGORIES: CategoryInput[]
```

### src/db/categoryStore.ts
```ts
import type { Category } from '../lib/category'
export function addCategory(c: Category): Promise<void>
export function getAllCategories(): Promise<Category[]>
export function updateCategory(c: Category): Promise<void>
export function removeCategory(id: string): Promise<void>
// Idempotent seed — if the categories store is empty, create + add one Category
// per DEFAULT_CATEGORIES (via createCategory). No-op if non-empty. Returns the
// resulting full list.
export function seedDefaultCategories(): Promise<Category[]>
```
Uses `withStore('categories', ...)` from `./db`. Type-only import of `Category`.
